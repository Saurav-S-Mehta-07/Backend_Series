## using wrapAsync
function asyncWrap(fn){
    return function(req,res,next){
        fn(req,res,next).catch(err => next(err));
    }
}

app.get("/chats/:id/edit", asyncWrap(async(req,res)=>{
    let {id} = req.params
    let chat = await Chat.findById(id)
    res.render("edit", {chat})
}))
