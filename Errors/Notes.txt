// Error-handling middleware MUST be last
app.use((err, req, res, next) => {
    console.log("-----------------error----------");
    next(err);
});

error handling middleware mai jab next() ko call lagate hai 
next will search for next non-error handling middleware
so we will pass err inside that too and thats how we trigger express's default
error handler otherwise it will show can't get /error_route

*****************************************************

app.get("/error", (req, res) => {
    abcd = abcd;
});

app.use((err, req, res, next) => {
    console.log("-----------------error----------");
    next();
});

app.use((err,req,res,next)=>{
    console.log("-------2nd error handler------");
    next(err);
})

// can't get /error

app.get("/error", (req, res) => {

app.use((err,req,res,next)=>{

next() or next(err)

next() : call next non-error handling middleware
next(err) : call next error handling middleware
